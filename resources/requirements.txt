# Dépendances Python du démon ProJote (installées dans resources/python_venv).
# Source de vérité : post-install.sh fait `pip install -r requirements.txt`.
# Versions relevées par gel du venv de production (python 3.11) — épinglées,
# sauf pronotepy : borne haute <3.0 pour suivre les correctifs 2.x sans subir
# un breaking change 3.x (décision v1.0.1).
#
# Attention aux épingles issues d'un gel sous python 3.11 : Jeedom tourne aussi
# sur Debian bullseye, qui fournit python 3.9 (cas courant sur Raspberry Pi).
# Une version exigeant python >= 3.10 y rend l'installation des dépendances
# impossible — pip échoue sur « Could not find a version that satisfies the
# requirement » et le plugin reste sans démon. Avant de relever une épingle,
# vérifier le champ « Requires-Python » de la version visée ; si elle exclut
# 3.9, la déclarer par marqueur d'environnement comme requests ci-dessous.
#
# Plancher 2.15.6 (obligatoire) : les serveurs PRONOTE 2026 ne publient plus
# l'appel Start({...}) dans l'attribut « onload » du <body>. pronotepy <= 2.14.6
# le lisait via body["onload"] et échoue donc sur « KeyError: 'onload' » à la
# connexion (identifiants comme QR Code). 2.15.6 cherche Start({...}) dans toute
# la page et envoie un User-Agent mobile pour les espaces mobile.*.html.
# ProJoted.py gère les deux API d'Information.content (propriété en 2.14,
# méthode en 2.15), la borne peut donc être relevée sans autre changement.
pronotepy>=2.15.6,<3.0
autoslot==2025.11.1
cryptography==46.0.6
pycryptodome==3.20.0
# requests 2.33 exige python >= 3.10 : 2.32.5 est la dernière compatible 3.9.
requests==2.33.0; python_version >= "3.10"
requests==2.32.5; python_version < "3.10"
beautifulsoup4==4.14.3
pyserial==3.5
pyudev==0.24.4
